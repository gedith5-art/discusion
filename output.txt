def _base_wheel_velocity_clip() -> dict[str, tuple[float, float]]:
    max_wheel_velocity = 0.45 / DEFAULT_BASE_WHEEL_RADIUS
    return {
        joint_name: (-max_wheel_velocity, max_wheel_velocity)
        for joint_name in G1_BASE_WHEEL_DOF_ORDER
    }


@configclass
class IoaiSimSceneFlowActionsCfg:
    """Union of existing G1 action groups required by all three phases."""

    base_action = g1_action_cfg("base", "velocity", clip=_base_wheel_velocity_clip())
    leg_action = g1_action_cfg("legs", "absolute")
    left_arm_action = g1_action_cfg("left_arm", "absolute")
    left_gripper_action = g1_action_cfg("left_gripper", "absolute")
    right_arm_action = g1_action_cfg("right_arm", "absolute")
    right_gripper_action = g1_action_cfg("right_gripper", "absolute")


def _planner_agent(task_id: str):
    def _factory(_env: Any):
        from ioailab.agents import CuroboPlannerAgent

        return CuroboPlannerAgent.from_task(task_id)

    return _factory


def _nav_agent(env: Any):
    unwrapped = getattr(env, "unwrapped", env)
    product_id = str(unwrapped.cfg.selected_ioai_product)
    goal_position = tuple(float(value) for value in unwrapped.cfg.goal_position)
    return nav_sequence_agent(
        product_id=str(product_id),
        goal_xy=(goal_position[0], goal_position[1]),
        goal_yaw=float(unwrapped.cfg.goal_yaw),
    )

def _apply_task_options(self, task_options: Mapping[str, object]) -> None:
    """Select one exact non-cocoa product for the coherent flow."""

    options = dict(task_options)
    defer_success = bool(options.pop("defer_success_termination", False))
    table_layout_path = options.pop("table_layout", None)
    product_id = options.pop("pick_product", IOAI_SIM_SCENE_FLOW_DEFAULT_PRODUCT)
    if options:
        raise ValueError(f"Unsupported IOAI flow task option(s): {sorted(options)}")
    if (
        not isinstance(product_id, str)
        or product_id not in IOAI_SIM_SCENE_FLOW_PRODUCT_IDS
    ):
        choices = ", ".join(IOAI_SIM_SCENE_FLOW_PRODUCT_IDS)
        raise ValueError(
            f"IOAI Pick→Nav→Place pick_product must be non-cocoa: {choices}."
        )
    product_index = IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID[product_id]
    pick_params = self.events.reset_all.params["pick_scene_params"]
    pick_params["target_product_index"] = product_index
    pick_params["zone_occupancy"] = IOAI_SIM_SCENE_PICK_OCCUPANCY_BY_TARGET_ZONE[
        IOAI_SIM_SCENE_PICK_ZONE_INDEX_BY_PRODUCT_INDEX[product_index]
    ]
    if table_layout_path is not None:
        if not isinstance(table_layout_path, str) or not table_layout_path.strip():
            raise ValueError("IOAI flow table_layout must be a non-empty path.")
        layout = load_ioai_table_layout(table_layout_path)
        pick_params["random_product_pools"] = (
            (IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID[layout.product_at("B1")],),
            (IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID[layout.product_at("B2")],),
        )
    self.selected_ioai_product = product_id
    if defer_success:
        self.terminations.success = None


GalbotG1IoaiSimSceneEnvCfg.apply_task_options = _apply_task_options
