    if args.num_envs != 1:
        parser.error("IOAI policy pipeline v1 requires --num-envs 1.")
    if args.episodes < 1 or args.max_steps < 1:
        parser.error("--episodes and --max-steps must be positive.")
    if args.policy_cache_size < 1:
        parser.error("--policy-cache-size must be positive.")
    try:
        limits = IoaiPipelineLimits(
            pick_max_steps=args.pick_max_steps,
            nav_max_steps=args.nav_max_steps,
            place_max_steps=args.place_max_steps,
            drop_confirm_steps=args.drop_confirm_steps,
            delivery_stable_steps=args.delivery_stable_steps,
        )
    except ValueError as exc:
        parser.error(str(exc))

    layout = load_ioai_table_layout(args.table_layout)
    work_order = layout.work_order(args.max_products)
    manifest = load_policy_manifest(
        args.policy_manifest,
        required_products=work_order,
    )
    logger.info(
        "IOAI table layout=%s work_order=%s",
        dict(zip(layout.slots, layout.products, strict=True)),
        work_order,
    )

    env = make_env(
        GALBOT_G1_IOAI_SIM_SCENE_TASK_ID,
        num_envs=1,
        headless=args.headless,
        enable_recorders=False,
        task_options={
            "pick_product": "gum",
            "table_layout": args.table_layout,
            "defer_success_termination": True,
        },
    )
    cache = PolicyAgentCache(args.policy_cache_size)
    agent = IoaiPolicyPipelineAgent(
        layout=layout,
        max_products=args.max_products,
        product_agent_factory=partial(
            _product_flow_agent,
            manifest=manifest,
            cache=cache,
        ),
        product_prefetcher=partial(
            _prefetch_next_pick_policy,
            manifest=manifest,
            cache=cache,
        ),
        limits=limits,
    )
    try:
        metrics = evaluate_policy_pipeline(
            env,
            agent,
            episodes=args.episodes,
            max_steps=args.max_steps,
            seed=args.seed,
        )
        logger.info(
            "IOAI policy baseline: %d/%d successful; work_order=%s lengths=%s",
            metrics["success_count"],
            metrics["total_episodes"],
            metrics["work_order"],
            metrics["episode_lengths"],
        )
        control_period = _control_period(env)
        total_steps = sum(metrics["episode_lengths"])
        logger.info(
            "IOAI policy baseline sim time: %.2fs (%d steps x %.4fs/step).",
            total_steps * control_period,
            total_steps,
            control_period,
        )
        for episode_index, failure in enumerate(metrics["failures"], start=1):
            if failure is not None:
                logger.error(
                    "IOAI policy baseline failure: episode=%d failure=%s",
                    episode_index,
                    failure,
                )
    finally:
        agent.close()
        cache.close()
        env.close()


if __name__ == "__main__":
    main()
