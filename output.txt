def read_table_layout(env: Any, *, env_id: int = 0) -> IoaiTableLayout:
    """Read the reset-time table layout snapshot without consulting live poses."""

    unwrapped = _unwrapped_env(env)
    device = getattr(unwrapped, "device", "cpu")
    num_envs = int(unwrapped.num_envs)
    row = int(env_id)
    if not 0 <= row < num_envs:
        raise ValueError(f"env_id {row} is outside [0, {num_envs}).")
    indices = table_layout_product_indices(
        unwrapped,
        num_envs=num_envs,
        product_count=len(IOAI_SIM_SCENE_PICK_PRODUCT_IDS),
        device=device,
    )[row]
    occupancy = table_layout_occupancy(
        unwrapped,
        num_envs=num_envs,
        device=device,
    )[row]
    product_by_index = {
        index: product_id
        for product_id, index in IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID.items()
    }
    return IoaiTableLayout(
        products=tuple(product_by_index[int(index)] for index in indices.tolist()),
        occupancy=tuple(bool(value) for value in occupancy.tolist()),
    )


def activate_pipeline_product(
    env: Any,
    product_id: str,
    *,
    env_ids: tuple[int, ...] = (0,),
) -> None:
    """Activate one live product by updating target state only.

    This operation deliberately does not write robot joints, grippers, or product
    root poses. The saved layout is also left untouched.
    """

    resolved = str(product_id)
    if resolved not in IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID:
        choices = ", ".join(IOAI_SIM_SCENE_PICK_PRODUCT_IDS)
        raise ValueError(
            f"Unknown IOAI product {resolved!r}; expected one of {choices}."
        )
    unwrapped = _unwrapped_env(env)
    rows = tuple(int(env_id) for env_id in env_ids)
    if not rows:
        raise ValueError("env_ids must not be empty.")
    for row in rows:
        layout = read_table_layout(unwrapped, env_id=row)
        matches = tuple(
            index
            for index, candidate in enumerate(layout.products)
            if candidate == resolved and layout.occupancy[index]
        )
        if len(matches) != 1:
            raise RuntimeError(
                f"Product {resolved!r} must occupy exactly one saved slot in env {row}."
            )

    device = getattr(unwrapped, "device", "cpu")
    ids = torch.tensor(rows, device=device, dtype=torch.long)
    product_index = IOAI_SIM_SCENE_PICK_PRODUCT_INDEX_BY_ID[resolved]
    selected = torch.full((len(rows),), product_index, device=device, dtype=torch.long)
    set_selected_product_indices(
        unwrapped,
        env_ids=ids,
        selected_product_indices=selected,
    )
    reset_params = unwrapped.cfg.events.reset_all.params
    pick_params = reset_params["pick_scene_params"]
    update_pick_target_from_current_product(
        unwrapped,
        ids,
        selected_product_indices=selected,
        product_cfgs=pick_params["product_cfgs"],
        pick_target_cfg=pick_params["pick_target_cfg"],
        product_geometry_centers=pick_params["product_geometry_centers"],
        product_pick_target_z_offsets=pick_params["product_pick_target_z_offsets"],
    )
    initialize_place_targets(
        unwrapped,
        ids,
        product_count=int(reset_params["product_count"]),
        shelf_layers=reset_params["shelf_layers"],
        shelf_position=reset_params["shelf_position"],
        shelf_orientation_xyzw=reset_params["shelf_orientation_xyzw"],
    )
    unwrapped.cfg.selected_ioai_product = resolved


def set_place_navigation_goal(
    env: Any,
    *,
    base_xy: Sequence[float],
    base_yaw: float,
) -> None:
    """Use a Place Scenario base pose as the live Nav target without teleporting."""

    coordinates = tuple(float(value) for value in base_xy)
    if len(coordinates) != 2:
        raise ValueError("Place navigation base_xy must contain exactly two values.")
    unwrapped = _unwrapped_env(env)
    cfg = unwrapped.cfg
    current_goal = tuple(float(value) for value in cfg.goal_position)
    goal_z = current_goal[2] if len(current_goal) >= 3 else 0.0
    cfg.goal_position = (coordinates[0], coordinates[1], goal_z)
    cfg.goal_yaw = float(base_yaw)

@dataclass(frozen=True, slots=True)
class IoaiPipelineLimits:
    """Bound one product attempt without terminating the whole work order."""

    pick_max_steps: int = 1200
    nav_max_steps: int = 1200
    place_max_steps: int = 1200
    drop_confirm_steps: int = 20
    delivery_stable_steps: int = 10

    def __post_init__(self) -> None:
        values = {
            "pick_max_steps": self.pick_max_steps,
            "nav_max_steps": self.nav_max_steps,
            "place_max_steps": self.place_max_steps,
            "drop_confirm_steps": self.drop_confirm_steps,
            "delivery_stable_steps": self.delivery_stable_steps,
        }
        invalid = tuple(name for name, value in values.items() if int(value) < 1)
        if invalid:
            raise ValueError(f"IOAI pipeline limits must be positive: {invalid}.")

    def max_steps(self, phase: str) -> int:
        """Return the configured budget for one canonical phase."""

        phase_name = str(phase)
        if phase_name not in IOAI_PIPELINE_PHASES:
            raise ValueError(f"Unknown IOAI pipeline phase {phase_name!r}.")
        return int(getattr(self, f"{phase_name}_max_steps"))


class _IoaiProductScore:
    """Mutable per-product score state owned by one episode scorecard."""

    def __init__(self, product_id: str) -> None:
        self.product_id = str(product_id)
        self.phase_status = {phase: "pending" for phase in IOAI_PIPELINE_PHASES}
        self.failure: dict[str, str] | None = None

    def mark_success(self, phase: str) -> None:
        phase_name = _validate_pipeline_phase(phase)
        if self.phase_status[phase_name] in {"failed", "skipped"}:
            raise RuntimeError(
                f"Cannot mark {self.product_id}/{phase_name} successful after failure."
            )
        self.phase_status[phase_name] = "success"

    def mark_failure(self, phase: str, reason: str) -> None:
        phase_name = _validate_pipeline_phase(phase)
        if self.failure is not None:
            return
        self.phase_status[phase_name] = "failed"
        phase_index = IOAI_PIPELINE_PHASES.index(phase_name)
        for skipped_phase in IOAI_PIPELINE_PHASES[phase_index + 1 :]:
            if self.phase_status[skipped_phase] == "pending":
                self.phase_status[skipped_phase] = "skipped"
        self.failure = {"phase": phase_name, "reason": str(reason)}

    def mark_complete(self) -> None:
        for phase in IOAI_PIPELINE_PHASES:
            self.mark_success(phase)

    @property
    def complete(self) -> bool:
        return all(
            self.phase_status[phase] == "success" for phase in IOAI_PIPELINE_PHASES
        )

    @property
    def total(self) -> int:
        phase_points = sum(
            IOAI_PIPELINE_PHASE_SCORE
            for phase in IOAI_PIPELINE_PHASES
            if self.phase_status[phase] == "success"
        )
        return phase_points + (IOAI_PIPELINE_COMPLETION_BONUS if self.complete else 0)

    def matrix_row(self) -> dict[str, Any]:
        status = "success" if self.complete else "failed" if self.failure else "pending"
        return {
            "product": self.product_id,
            **{
                phase: (
                    IOAI_PIPELINE_PHASE_SCORE
                    if self.phase_status[phase] == "success"
                    else 0
                )
                for phase in IOAI_PIPELINE_PHASES
            },
            "bonus": IOAI_PIPELINE_COMPLETION_BONUS if self.complete else 0,
            "total": self.total,
            "status": status,
            "failure": None if self.failure is None else dict(self.failure),
        }
