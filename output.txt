@dataclass(frozen=True, slots=True)
class IoaiPipelineLimits:
    """Bound one product attempt without terminating the whole work order."""

    pick_max_steps: int = 1200
    nav_max_steps: int = 1200
    place_max_steps: int = 1200
    drop_confirm_steps: int = 20
    delivery_stable_steps: int = 10

    def __post_init__(self) -> None:
        values = {
            "pick_max_steps": self.pick_max_steps,
            "nav_max_steps": self.nav_max_steps,
            "place_max_steps": self.place_max_steps,
            "drop_confirm_steps": self.drop_confirm_steps,
            "delivery_stable_steps": self.delivery_stable_steps,
        }
        invalid = tuple(name for name, value in values.items() if int(value) < 1)
        if invalid:
            raise ValueError(f"IOAI pipeline limits must be positive: {invalid}.")

    def max_steps(self, phase: str) -> int:
        """Return the configured budget for one canonical phase."""

        phase_name = str(phase)
        if phase_name not in IOAI_PIPELINE_PHASES:
            raise ValueError(f"Unknown IOAI pipeline phase {phase_name!r}.")
        return int(getattr(self, f"{phase_name}_max_steps"))


class _IoaiProductScore:
    """Mutable per-product score state owned by one episode scorecard."""

    def __init__(self, product_id: str) -> None:
        self.product_id = str(product_id)
        self.phase_status = {phase: "pending" for phase in IOAI_PIPELINE_PHASES}
        self.failure: dict[str, str] | None = None

    def mark_success(self, phase: str) -> None:
        phase_name = _validate_pipeline_phase(phase)
        if self.phase_status[phase_name] in {"failed", "skipped"}:
            raise RuntimeError(
                f"Cannot mark {self.product_id}/{phase_name} successful after failure."
            )
        self.phase_status[phase_name] = "success"

    def mark_failure(self, phase: str, reason: str) -> None:
        phase_name = _validate_pipeline_phase(phase)
        if self.failure is not None:
            return
        self.phase_status[phase_name] = "failed"
        phase_index = IOAI_PIPELINE_PHASES.index(phase_name)
        for skipped_phase in IOAI_PIPELINE_PHASES[phase_index + 1 :]:
            if self.phase_status[skipped_phase] == "pending":
                self.phase_status[skipped_phase] = "skipped"
        self.failure = {"phase": phase_name, "reason": str(reason)}

    def mark_complete(self) -> None:
        for phase in IOAI_PIPELINE_PHASES:
            self.mark_success(phase)

    @property
    def complete(self) -> bool:
        return all(
            self.phase_status[phase] == "success" for phase in IOAI_PIPELINE_PHASES
        )

    @property
    def total(self) -> int:
        phase_points = sum(
            IOAI_PIPELINE_PHASE_SCORE
            for phase in IOAI_PIPELINE_PHASES
            if self.phase_status[phase] == "success"
        )
        return phase_points + (IOAI_PIPELINE_COMPLETION_BONUS if self.complete else 0)

    def matrix_row(self) -> dict[str, Any]:
        status = "success" if self.complete else "failed" if self.failure else "pending"
        return {
            "product": self.product_id,
            **{
                phase: (
                    IOAI_PIPELINE_PHASE_SCORE
                    if self.phase_status[phase] == "success"
                    else 0
                )
                for phase in IOAI_PIPELINE_PHASES
            },
            "bonus": IOAI_PIPELINE_COMPLETION_BONUS if self.complete else 0,
            "total": self.total,
            "status": status,
            "failure": None if self.failure is None else dict(self.failure),
        }


class IoaiPipelineScorecard:
    """Five-point phase matrix and completion bonus for one pipeline episode."""

    def __init__(self, work_order: Sequence[str]) -> None:
        products = tuple(str(product_id) for product_id in work_order)
        if not products or len(set(products)) != len(products):
            raise ValueError("IOAI scorecard requires a non-empty unique work order.")
        self.work_order = products
        self._results: dict[str, _IoaiProductScore] = {}
        self.reset()

    def reset(self) -> None:
        """Clear results for a new environment reset."""

        self._results = {
            product_id: _IoaiProductScore(product_id) for product_id in self.work_order
        }

    def mark_phase_success(self, product_id: str, phase: str) -> None:
        self._result(product_id).mark_success(phase)
