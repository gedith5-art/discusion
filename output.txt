"""Run one IOAI Pick→Nav→Place pipeline and capture the Place start."""

from __future__ import annotations

import argparse
from typing import Any

from ioailab.agents import PolicyAgent, TaskFlowAgent
from ioailab.envs import make_env
from ioailab.tasks.ioai_sim_scene.scenario_capture import (
    PlaceStartScenarioCaptureAgent,
)
from ioailab.tasks.ioai_sim_scene.scene import IOAI_SIM_SCENE_FLOW_PRODUCT_IDS
from ioailab.utils.log_utils import configure, get_logger

logger = get_logger(__name__)

TASK_ID = "GalbotG1-IOAISimScene-v0"


def _evaluate_until_scenario_count(
    env: Any,
    agent: PlaceStartScenarioCaptureAgent,
    *,
    target_scenarios: int,
    max_attempts: int,
    max_steps: int,
    product: str,
) -> dict[str, int | float]:
    """Evaluate batches until the requested number of Scenarios is captured."""

    if target_scenarios < 1:
        raise ValueError("target_scenarios must be greater than zero.")
    if max_attempts < target_scenarios:
        raise ValueError("max_attempts must be at least target_scenarios.")

    attempts = 0
    success_count = 0
    while len(agent.saved_paths) < target_scenarios:
        remaining_attempts = max_attempts - attempts
        if remaining_attempts <= 0:
            raise RuntimeError(
                f"Scenario collection captured {len(agent.saved_paths)} of "
                f"{target_scenarios} {product} Place starts after {attempts} attempts."
            )
        batch_size = min(
            target_scenarios - len(agent.saved_paths),
            remaining_attempts,
        )
        metrics = env.evaluate(
            agent=agent,
            episodes=batch_size,
            max_steps=max_steps,
        )
        evaluated = int(metrics["total_episodes"])
        if evaluated != batch_size:
            raise RuntimeError(
                f"Expected evaluation batch size {batch_size}, got {evaluated}."
            )
        attempts += evaluated
        success_count += int(metrics["success_count"])
        logger.info(
            "Captured %s/%s %s Place start Scenario(s) after %s attempt(s)",
            len(agent.saved_paths),
            target_scenarios,
            product,
            attempts,
        )

    if len(agent.saved_paths) != target_scenarios:
        raise RuntimeError(
            f"Scenario collection exceeded target {target_scenarios}: "
            f"captured {len(agent.saved_paths)}."
        )
    return {
        "total_episodes": attempts,
        "success_count": success_count,
        "success_rate": success_count / float(attempts),
    }


def main(argv: list[str] | None = None) -> None:
    configure()
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--product",
        choices=IOAI_SIM_SCENE_FLOW_PRODUCT_IDS,
        required=True,
        help="Exact non-cocoa product to run through the pipeline.",
    )
    parser.add_argument(
        "--pick-checkpoint",
        default=None,
        help="Optional Pick policy checkpoint; Nav and Place remain task defaults.",
    )
    parser.add_argument(
        "--dataset-path",
        default=None,
        help="Optional full-pipeline HDF5 output path; omit to save only Scenario.",
    )
    scenario_output = parser.add_mutually_exclusive_group()
    scenario_output.add_argument(
        "--save-place-scenario",
        default=None,
        help=(
            "Nav→Place Scenario YAML path. Defaults to "
            "data/ioai_sim_scene_place/scenarios/<product>.yaml."
        ),
    )
    scenario_output.add_argument(
        "--save-place-scenario-dir",
        default=None,
        help="Directory for numbered Nav→Place Scenarios: 000.yaml, 001.yaml, ...",
    )
    parser.add_argument("--episodes", type=int, default=1)
    parser.add_argument(
        "--max-scenario-attempts",
        type=int,
        default=None,
        help=(
            "Maximum full-pipeline attempts when collecting a Scenario directory; "
            "defaults to 10 times --episodes."
        ),
    )
    parser.add_argument("--num-envs", type=int, default=1)
    parser.add_argument("--max-steps", type=int, default=3000)
    parser.add_argument("--headless", action="store_true")
    args = parser.parse_args(argv)

    if args.save_place_scenario_dir is not None and args.num_envs != 1:
        raise ValueError("--save-place-scenario-dir requires --num-envs 1.")
    if args.max_scenario_attempts is not None and (
        args.save_place_scenario_dir is None or args.dataset_path is not None
    ):
        raise ValueError(
            "--max-scenario-attempts requires Scenario-only "
            "--save-place-scenario-dir collection."
        )

    scenario_path = None
    if args.save_place_scenario_dir is None:
        scenario_path = args.save_place_scenario or (
            f"data/ioai_sim_scene_place/scenarios/{args.product}.yaml"
        )
    env = make_env(
        TASK_ID,
        num_envs=args.num_envs,
        headless=args.headless,
        enable_recorders=args.dataset_path is not None,
        task_options={"pick_product": args.product},
    )
    task_flow_agent = (
        TaskFlowAgent.from_env(env)
        if args.pick_checkpoint is None
        else TaskFlowAgent.from_env(
            env,
            agents={"pick": PolicyAgent.from_checkpoint(args.pick_checkpoint)},
        )
    )
    agent = PlaceStartScenarioCaptureAgent(
        task_flow_agent,
        product=args.product,
        path=scenario_path,
        directory=args.save_place_scenario_dir,
    )
    try:
        if args.dataset_path is None:
            if args.save_place_scenario_dir is None:
                metrics = env.evaluate(
                    agent=agent,
                    episodes=args.episodes,
                    max_steps=args.max_steps,
                )
            else:
                max_attempts = args.max_scenario_attempts or args.episodes * 10
                metrics = _evaluate_until_scenario_count(
                    env,
                    agent,
                    target_scenarios=args.episodes,
                    max_attempts=max_attempts,
                    max_steps=args.max_steps,
                    product=args.product,
                )
            logger.info(
                "Ran %s full-pipeline %s attempt(s) without saving HDF5 "
                "(full-task success rate %.1f%%)",
                metrics["total_episodes"],
                args.product,
                metrics["success_rate"] * 100.0,
            )
        else:
            dataset = env.collect(
                agent=agent,
                path=args.dataset_path,
                episodes=args.episodes,
                max_steps=args.max_steps,
            )
            logger.info(
                "Saved %s full-pipeline %s episode(s) to %s",
                args.episodes,
                args.product,
                dataset.path,
            )
        if not agent.saved_paths:
            raise RuntimeError(
                "Pipeline did not reach the Nav→Place transition; no Place "
                "Scenario was saved."
            )
        if (
            args.save_place_scenario_dir is not None
            and len(agent.saved_paths) != args.episodes
        ):
            raise RuntimeError(
                f"Expected {args.episodes} Place start Scenarios, but captured "
                f"{len(agent.saved_paths)}."
            )
        logger.info(
            "Saved %s %s Place start Scenario(s): %s",
            len(agent.saved_paths),
            args.product,
            agent.saved_paths,
        )
    finally:
        env.close()


if __name__ == "__main__":

def _write_checkpoint(directory: Path, task_id: str) -> Path:
    directory.mkdir(parents=True)
    checkpoint = directory / "model_inference.pth"
    checkpoint.touch()
    (directory / "config.json").write_text(
        json.dumps({"experiment": {"name": task_id}}),
        encoding="utf-8",
    )
    (directory / "action_norm_params.json").write_text(
        json.dumps({"action_shape": [16]}),
        encoding="utf-8",
    )
    return checkpoint


def _write_scenario(path: Path, product: str) -> Path:
    path.write_text(
        yaml.safe_dump(
            {
                "schema": "ioailabScenario-v0",
                "frame": "env",
                "metadata": {
                    "task_id": "GalbotG1-IOAISimScene-v0",
                    "kind": "place_start",
                    "product": product,
                    "transition": "nav_to_place",
                },
                "assets": {
                    "articulation": {
                        "robot": {
                            "base_pose": [
                                0.51,
                                -0.02,
                                0.0,
                                0.0,
                                0.0,
                                0.01499943750632809,
                                0.9998875021093592,
                            ],
                            "joint_position": {
                                joint_name: float(index + 1)
                                for index, joint_name in enumerate(
                                    (
                                        *G1_BASE_WHEEL_DOF_ORDER,
                                        *G1_ASSET_INTERNAL_DRIVEN_JOINT_NAMES,
                                        *IOAI_SIM_SCENE_PLACE_PHYSICAL_JOINT_NAMES,
                                        *G1_LEFT_GRIPPER_DOF_ORDER,
                                        *G1_RIGHT_GRIPPER_DOF_ORDER,
                                    )
                                )
                            },
                        }
                    }
                },
            }
        ),
        encoding="utf-8",
    )
    return path
